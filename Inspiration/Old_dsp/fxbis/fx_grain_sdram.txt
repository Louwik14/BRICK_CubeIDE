
#include "fx_granular.h"
#include "sdram.h"

namespace {

constexpr uint32_t kBufferSize = 16000u;
constexpr uint32_t kMaxGrains = 10u;
constexpr uint32_t kMinDurationSamples = 480u;
constexpr uint32_t kMaxDurationSamples = 5760u;

// ===== D1 CACHE (ring) =====
// Choix: 2048 samples (≈ 42.7ms @48k) -> spread "réel" sans SDRAM random.
// Power of two obligatoire.
constexpr uint32_t kCacheSize = 8192u;
constexpr uint32_t kCacheMask = kCacheSize - 1u;
constexpr uint32_t kCacheLimit = kCacheSize << 16;

// D1 RAM (ne surtout pas mettre en SDRAM_BSS)
alignas(32) static float g_cache_l[kCacheSize];
alignas(32) static float g_cache_r[kCacheSize];

struct Grain {
  uint32_t pos;        // 16.16 in CACHE space
  uint32_t increment;  // 16.16 step in CACHE space
  float gain_l;
  float gain_r;
  float env;
  float env_inc;
  uint32_t remaining;
  bool active;

  uint8_t interp_toggle;
  float lastL;
  float lastR;
  float holdL;   // NEW
  float holdR;   // NEW
};

struct GranularState {
  float sample_rate;
  float density;
  float pitch_semitones;
  float pitch_increment;
  float mix;
  bool freeze;
  float spread;
  float stereo_offset;
  float* buffer_l;
  float* buffer_r;
  uint32_t write_pos;     // SDRAM write head

  // D1 cache write head
  uint32_t cache_wp;      // 0..kCacheSize-1

  Grain grains[kMaxGrains];

  uint32_t rng;

  uint32_t spawn_counter;
  uint32_t spawn_interval;

  uint32_t free_index[kMaxGrains];
  uint32_t free_count;
};

alignas(4) SDRAM_BSS float g_granular_buffer_l[kBufferSize];
alignas(4) SDRAM_BSS float g_granular_buffer_r[kBufferSize];

GranularState g_state;

constexpr float kInvNorm[] = {
    1.0000f, 1.0000f, 0.9000f, 0.8300f, 0.7700f, 0.7100f, 0.6700f,
    0.6300f, 0.6000f, 0.5700f, 0.5500f, 0.5300f, 0.5100f, 0.4950f,
    0.4800f, 0.4650f, 0.4500f, 0.4350f, 0.4200f, 0.4100f, 0.4000f
};

inline float clamp(float x, float lo, float hi) {
  if (x < lo) return lo;
  if (x > hi) return hi;
  return x;
}

inline uint32_t next_rand() {
  uint32_t x = g_state.rng;
  x ^= x << 13;
  x ^= x >> 17;
  x ^= x << 5;
  g_state.rng = x;
  return x;
}

inline float rand_0_1() {
  return (next_rand() & 0x00FFFFFFu) * (1.0f / 16777216.0f);
}

inline float fast_exp2_approx(float x) {
  const float x2 = x * x;
  const float x3 = x2 * x;
  return 1.0f + (0.69314718f * x) + (0.24022651f * x2) + (0.05550411f * x3);
}

inline void update_pitch_increment() {
  const float semitones = clamp(g_state.pitch_semitones, -48.0f, 48.0f);
  const float octaves = semitones * (1.0f / 12.0f);
  float ratio = fast_exp2_approx(octaves);
  if (ratio < 0.0625f) ratio = 0.0625f;
  if (ratio > 16.0f) ratio = 16.0f;
  g_state.pitch_increment = ratio * 65536.0f;
}

inline void update_spawn_probability() {
  float d = clamp(g_state.density, 0.0f, 1.0f);
  float grains_per_sec = 5.0f + d * 120.0f;

  g_state.spawn_interval = (uint32_t)(g_state.sample_rate / grains_per_sec);
  if (g_state.spawn_interval < 1) g_state.spawn_interval = 1;
}

// Interp dans le CACHE D1 (pas SDRAM)
inline float read_cache_interp(const float* buf, uint32_t pos) {
  uint32_t i0 = (pos >> 16) & kCacheMask;
  uint32_t i1 = (i0 + 1u) & kCacheMask;

  float a = buf[i0];
  float b = buf[i1];

  float frac = (pos & 0xFFFFu) * (1.0f / 65536.0f);
  return a + (b - a) * frac;
}

inline void spawn_grain() {
  if (g_state.free_count == 0) return;

  uint32_t idx = g_state.free_index[--g_state.free_count];
  Grain& g = g_state.grains[idx];

  // ✅ SPREAD ORIGINAL (comme ton ancien algo), mais dans le ring D1
  uint32_t max_span = (uint32_t)(g_state.spread * (float)kCacheSize);
  if (max_span < 32u) max_span = 32u;
  if (max_span > (kCacheSize - 1u)) max_span = (kCacheSize - 1u);

  float r = rand_0_1();
  r = r * r; // bias vers centre
  uint32_t rand_span = (uint32_t)(r * (float)max_span);

  // start = cache_wp - rand_span (dans le passé)
  uint32_t start = (g_state.cache_wp - rand_span) & kCacheMask;

  float pan = rand_0_1();

  uint32_t duration = kMinDurationSamples +
    (next_rand() % (kMaxDurationSamples - kMinDurationSamples));

  g.pos = start << 16;

  // ✅ pas de detune "haunting"
  g.increment = (uint32_t)g_state.pitch_increment;

  // ✅ clamp léger (optionnel mais safe)
  const uint32_t min_inc = (1u << 14);  // ~0.25x
  const uint32_t max_inc = (4u << 16);  // 4.0x
  if (g.increment < min_inc) g.increment = min_inc;
  if (g.increment > max_inc) g.increment = max_inc;

  g.gain_l = 1.0f - pan;
  g.gain_r = pan;
  g.env = 0.0f;
  g.remaining = duration;

  uint32_t half = duration >> 1;
  g.env_inc = (half > 0u) ? (1.0f / (float)half) : 1.0f;

  float jitter = (rand_0_1() - 0.5f) * 0.08f;
  g.env_inc *= (1.0f + jitter);

  g.interp_toggle = 0;
  g.lastL = 0.0f;
  g.lastR = 0.0f;

  g.holdL = 0.0f;   // NEW
  g.holdR = 0.0f;   // NEW

  g.active = true;
}
} // namespace

extern "C" void fx_granular_set_density(float density_0_1) {
  g_state.density = clamp(density_0_1, 0.0f, 1.0f);
  update_spawn_probability();
}

extern "C" void fx_granular_set_pitch(float semitones_m48_p48) {
  g_state.pitch_semitones = clamp(semitones_m48_p48, -48.0f, 48.0f);
  update_pitch_increment();
}

extern "C" void fx_granular_set_freeze(bool freeze) {
  g_state.freeze = freeze;
}

extern "C" void fx_granular_set_mix(float mix_0_1) {
  g_state.mix = clamp(mix_0_1, 0.0f, 1.0f);
}

extern "C" void fx_granular_set_spread(float spread_0_1) {
  g_state.spread = clamp(spread_0_1, 0.0f, 1.0f);
}

extern "C" void fx_granular_set_stereo_offset(float amount_0_1) {
  g_state.stereo_offset = clamp(amount_0_1, 0.0f, 1.0f);
}

extern "C" void fx_granular_init(float sample_rate) {
  g_state.sample_rate = sample_rate;
  g_state.density = 0.5f;
  g_state.pitch_semitones = 0.0f;
  g_state.mix = 1.0f;
  g_state.freeze = false;
  g_state.spread = 0.5f;
  g_state.stereo_offset = 0.5f;
  g_state.buffer_l = g_granular_buffer_l;
  g_state.buffer_r = g_granular_buffer_r;
  g_state.write_pos = 0u;

  g_state.cache_wp = 0u;

  g_state.rng = 0x12345678u;
  g_state.spawn_counter = 0;

  g_state.free_count = kMaxGrains;
  for (uint32_t i = 0; i < kMaxGrains; ++i) {
    g_state.grains[i].active = false;
    g_state.free_index[i] = i;
  }

  for (uint32_t i = 0; i < kBufferSize; ++i) {
    g_state.buffer_l[i] = 0.0f;
    g_state.buffer_r[i] = 0.0f;
  }

  for (uint32_t i = 0; i < kCacheSize; ++i) {
    g_cache_l[i] = 0.0f;
    g_cache_r[i] = 0.0f;
  }

  update_pitch_increment();
  update_spawn_probability();
}

extern "C" void fx_granular_process_block(float* in_l, float* in_r,
                                          float* out_l, float* out_r,
                                          uint32_t frames) {
  if (!in_l || !in_r || !out_l || !out_r) return;

  const float wet_mix = g_state.mix;
  const float dry_mix = 1.0f - wet_mix;

  float* sdramL = g_state.buffer_l;
  float* sdramR = g_state.buffer_r;

  // stereo offset en samples dans le cache (0..kCacheSize-1)
  uint32_t stereo_off = (uint32_t)(g_state.stereo_offset * (float)(kCacheSize - 1u));
  if (stereo_off > (kCacheSize - 1u)) stereo_off = (kCacheSize - 1u);

  for (uint32_t n = 0; n < frames; ++n) {
    const float inL = in_l[n];
    const float inR = in_r[n];

    if (!g_state.freeze) {
      // write SDRAM (record long)
      sdramL[g_state.write_pos] = inL;
      sdramR[g_state.write_pos] = inR;
      if (++g_state.write_pos >= kBufferSize) g_state.write_pos = 0u;

      // write D1 cache ring (read by grains)
      g_cache_l[g_state.cache_wp] = inL;
      g_cache_r[g_state.cache_wp] = inR;
      g_state.cache_wp = (g_state.cache_wp + 1u) & kCacheMask;
    }

    if (++g_state.spawn_counter >= g_state.spawn_interval) {
      g_state.spawn_counter = 0;
      spawn_grain();
    }

    float wetL = 0.0f;
    float wetR = 0.0f;
    uint32_t active_count = 0u;

    for (uint32_t i = 0; i < kMaxGrains; ++i) {
      Grain& g = g_state.grains[i];
      if (!g.active) continue;

      // 1 sample /2 (comme ton design low-cost)
      if (g.interp_toggle == 0) {
        // L interpolé dans cache D1
        g.lastL = read_cache_interp(g_cache_l, g.pos);

        // R: même position + offset, sans interpolation (cheap)
        uint32_t posR = g.pos + (stereo_off << 16);
        if (posR >= kCacheLimit) posR -= kCacheLimit;
        g.lastR = g_cache_r[(posR >> 16) & kCacheMask];
      }
      g.interp_toggle ^= 1;

      // NEW: smoothing du sample&hold (réduit les artefacts quand SRR monte)
      constexpr float kSlew = 0.35f;
      g.holdL += (g.lastL - g.holdL) * kSlew;
      g.holdR += (g.lastR - g.holdR) * kSlew;

      const float sL = g.holdL;
      const float sR = g.holdR;

      const float x = g.env;
      const float env = x * (2.0f - x);

      wetL += sL * env * g.gain_l;
      wetR += sR * env * g.gain_r;
      ++active_count;

      g.pos += g.increment;
      if (g.pos >= kCacheLimit) g.pos -= kCacheLimit;

      g.env += g.env_inc;

      if (--g.remaining == 0u || g.env >= 2.0f) {
        g.active = false;
        g_state.free_index[g_state.free_count++] = i;
      }
    }

    const float norm = kInvNorm[active_count];
    wetL *= norm;
    wetR *= norm;

    out_l[n] = inL * dry_mix + wetL * wet_mix;
    out_r[n] = inR * dry_mix + wetR * wet_mix;
  }
}
